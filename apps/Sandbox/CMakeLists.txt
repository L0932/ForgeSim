add_executable(
	ForgeSimSandbox 
)

set(
    FORGESIM_SANDBOX_GENERATED_INCLUDE_DIRECTORY
    "${CMAKE_CURRENT_BINARY_DIR}/generated"
)

configure_file(
    "${PROJECT_SOURCE_DIR}/cmake/Version.hpp.in"
    "${FORGESIM_SANDBOX_GENERATED_INCLUDE_DIRECTORY}/ForgeSim/Build/Version.hpp"
    @ONLY
)

target_include_directories(
    ForgeSimSandbox
    PRIVATE
        "${FORGESIM_SANDBOX_GENERATED_INCLUDE_DIRECTORY}"
)

add_custom_target(
	ForgeSimSandboxAssets
	ALL
	COMMAND
		${CMAKE_COMMAND} -E copy_directory
		"${CMAKE_CURRENT_SOURCE_DIR}/assets"
		"$<TARGET_FILE_DIR:ForgeSimSandbox>/assets"
	DEPENDS
		"${CMAKE_CURRENT_SOURCE_DIR}/assets/shaders/Grid.vert"
		"${CMAKE_CURRENT_SOURCE_DIR}/assets/shaders/Grid.frag"
		"${CMAKE_CURRENT_SOURCE_DIR}/assets/shaders/Object.vert"
		"${CMAKE_CURRENT_SOURCE_DIR}/assets/shaders/Object.frag"
        "${CMAKE_CURRENT_SOURCE_DIR}/assets/shaders/DirectionalShadow.vert"
        "${CMAKE_CURRENT_SOURCE_DIR}/assets/shaders/DirectionalShadow.frag"
		"${CMAKE_CURRENT_SOURCE_DIR}/assets/models/BoxVertexColors.glb"
		"${CMAKE_CURRENT_SOURCE_DIR}/assets/models/README.md"
	COMMENT
		"Copying Sandbox assets"
	VERBATIM
)

add_dependencies(
	ForgeSimSandbox
	ForgeSimSandboxAssets
)

target_compile_definitions(
	ForgeSimSandbox
	PRIVATE
	    GLM_ENABLE_EXPERIMENTAL
)

target_sources(
	ForgeSimSandbox
	PRIVATE
		Main.cpp
		EditorSession.hpp
        EditorSession.cpp
		EditorWorkspace.hpp
		EditorWorkspace.cpp
		LightGizmoTransform.hpp
		LightGizmoTransform.cpp
		SandboxApplication.hpp
		SandboxApplication.cpp
		SandboxDebugUI.hpp
		SandboxDebugUI.cpp
		SandboxObject.hpp
		SandboxPaths.hpp
		SandboxPaths.cpp
		SandboxRenderer.hpp
		SandboxRenderer.cpp
		SandboxScene.hpp
		SandboxScene.cpp
		SceneBounds.hpp
        SceneBounds.cpp
		SceneCameraData.hpp
		SceneCameraFraming.hpp
		SceneCameraFraming.cpp
		SceneCameraValidation.hpp
		SceneCameraValidation.cpp
		SceneEditHistory.hpp
		SceneEditHistory.cpp
		SceneFilePanel.hpp
        SceneFilePanel.cpp
		SceneHierarchyPanel.hpp
		SceneHierarchyPanel.cpp
		SceneInspectorPanel.hpp
		SceneInspectorPanel.cpp
		SceneLightEditing.hpp
		SceneLightEditing.cpp
		SceneLightValidation.hpp
		SceneLightValidation.cpp
		SceneTransformEditor.hpp
		SceneTransformEditor.cpp
		SceneViewportPanel.hpp
		SceneViewportPanel.cpp
		ShaderFileMonitor.hpp
		ShaderFileMonitor.cpp
		FreeCameraController.hpp
		FreeCameraController.cpp
		GizmoTransform.hpp
		GizmoTransform.cpp
		LightSubmission.hpp
		LightSubmission.cpp
		SandboxSceneDocument.cpp
		SceneDocument.hpp
		SceneDocumentIO.hpp
        SceneDocumentIO.cpp
		SceneDocumentValidation.hpp
        SceneDocumentValidation.cpp
		SceneSerializer.hpp
        SceneSerializer.cpp
		Transform.hpp
		Transform.cpp
		ViewportSelectionController.hpp
		ViewportSelectionController.cpp
		ViewportTransformGizmo.hpp
		ViewportTransformGizmo.cpp
)

target_compile_features(
	ForgeSimSandbox 
	PRIVATE
		cxx_std_20
)

target_link_libraries(
	ForgeSimSandbox
	PRIVATE
		ForgeSim::Assets
		ForgeSim::Core
		ForgeSim::PlatformGLFW
		ForgeSim::Renderer
		ForgeSim::RendererOpenGL
		ForgeSim::Glad
		ForgeSim::ImGui
		ForgeSim::ImGuizmo
		glm::glm-header-only
		nlohmann_json::nlohmann_json
)

forgesim_set_project_warnings(ForgeSimSandbox)

include("${PROJECT_SOURCE_DIR}/cmake/InstallSandbox.cmake")
